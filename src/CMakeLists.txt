# Adopted Kai engine; training/export/reference tools remain in the Kai repository.
set(KAI_SOURCES kai/model.cpp kai/cpu.cpp kai/cpu_optimized.cpp)
if(AGENTARMOR_METAL)
  list(APPEND KAI_SOURCES kai/metal.mm kai/metal_optimized.mm)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/kai/metal/kernels.metal" KAI_LEGACY_SHADER_SOURCE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/kai/metal/kernels.metal")
  set(KAI_OPT_SHADER_SOURCE "")
  foreach(shader common gemm elementwise attention)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/kai/metal/${shader}.metal" shader_source)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/kai/metal/${shader}.metal")
    string(APPEND KAI_OPT_SHADER_SOURCE "${shader_source}\n")
  endforeach()
  set(KAI_TENSOR_SHADER_SOURCE "")
  foreach(shader tensor_gemm tensor_attention)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/kai/metal/${shader}.metal" shader_source)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/kai/metal/${shader}.metal")
    string(APPEND KAI_TENSOR_SHADER_SOURCE "${shader_source}\n")
  endforeach()
  configure_file(kai/metal/shaders.h.in generated/shaders.h @ONLY)
endif()

add_library(armor_kai STATIC ${KAI_SOURCES})
target_include_directories(armor_kai PUBLIC kai PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(armor_kai SYSTEM PUBLIC "${PROJECT_SOURCE_DIR}/third_party")
target_link_libraries(armor_kai PUBLIC armor_build_options "-framework Foundation" "-framework Accelerate")
target_compile_definitions(armor_kai PRIVATE KAI_ACCELERATE=1 ACCELERATE_NEW_LAPACK=1)
target_compile_options(armor_kai PRIVATE -fblocks $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
if(AGENTARMOR_METAL)
  target_compile_definitions(armor_kai PUBLIC KAI_METAL=1)
  target_link_libraries(armor_kai PUBLIC "-framework Metal")
endif()
add_subdirectory(server)
