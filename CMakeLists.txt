cmake_minimum_required(VERSION 3.24)
project(AgentArmor VERSION 0.1.0 LANGUAGES CXX)

if(NOT APPLE)
  message(FATAL_ERROR "AgentArmor initially targets macOS on Apple Silicon M4 or newer")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  message(FATAL_ERROR "Use a native Apple Silicon toolchain, not Rosetta/x86_64")
endif()
if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS version" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(AGENTARMOR_METAL "Build Apple Metal engines" ON)
option(AGENTARMOR_BUILD_TESTS "Build weight-free security and contract tests" ON)
option(AGENTARMOR_BUILD_FUZZERS "Build Clang libFuzzer targets" OFF)
set(AGENTARMOR_SANITIZER "none" CACHE STRING "none, address (with UBSan), or thread")
set_property(CACHE AGENTARMOR_SANITIZER PROPERTY STRINGS none address thread)

add_library(armor_build_options INTERFACE)
target_compile_options(armor_build_options INTERFACE -Wall -Wextra -Wpedantic -Werror=return-type
  -fstack-protector-strong)
target_compile_definitions(armor_build_options INTERFACE
  _LIBCPP_HARDENING_MODE=_LIBCPP_HARDENING_MODE_EXTENSIVE)
if(AGENTARMOR_SANITIZER STREQUAL "address")
  target_compile_options(armor_build_options INTERFACE -fsanitize=address,undefined
    -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(armor_build_options INTERFACE -fsanitize=address,undefined)
elseif(AGENTARMOR_SANITIZER STREQUAL "thread")
  target_compile_options(armor_build_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(armor_build_options INTERFACE -fsanitize=thread)
elseif(NOT AGENTARMOR_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "AGENTARMOR_SANITIZER must be none, address, or thread")
endif()
if(AGENTARMOR_BUILD_TESTS)
  enable_testing()
endif()
add_subdirectory(native/kai)

install(TARGETS agentarmor RUNTIME DESTINATION bin)
install(FILES LICENSE NOTICE DESTINATION share/agentarmor COMPONENT licenses)
install(FILES native/kai/NOTICE DESTINATION share/agentarmor RENAME Kai-NOTICE COMPONENT licenses)
install(FILES native/kai/server/third_party/httplib.LICENSE native/kai/server/third_party/json.LICENSE
  DESTINATION share/agentarmor COMPONENT licenses)
