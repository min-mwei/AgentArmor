# Native Kai snapshot; trainer/reference tools remain in the Kai repository.
set(SOURCES src/model.cpp src/cpu.cpp src/cpu_optimized.cpp)
if(AGENTARMOR_METAL)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  list(APPEND SOURCES src/metal.mm src/metal_optimized.mm)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/metal/kernels.metal" KAI_LEGACY_SHADER_SOURCE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/metal/kernels.metal")
  set(KAI_OPT_SHADER_SOURCE "")
  foreach(shader common gemm elementwise attention)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/metal/${shader}.metal" shader_source)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/src/metal/${shader}.metal")
    string(APPEND KAI_OPT_SHADER_SOURCE "${shader_source}\n")
  endforeach()
  set(KAI_TENSOR_SHADER_SOURCE "")
  foreach(shader tensor_gemm tensor_attention)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/metal/${shader}.metal" shader_source)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
      "${CMAKE_CURRENT_SOURCE_DIR}/src/metal/${shader}.metal")
    string(APPEND KAI_TENSOR_SHADER_SOURCE "${shader_source}\n")
  endforeach()
  configure_file(src/metal/shaders.h.in generated/shaders.h @ONLY)
endif()

add_library(kaiinfer_core STATIC ${SOURCES})
target_include_directories(kaiinfer_core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(kaiinfer_core PUBLIC armor_build_options "-framework Foundation" "-framework Accelerate")
target_compile_definitions(kaiinfer_core PRIVATE KAI_ACCELERATE=1 ACCELERATE_NEW_LAPACK=1)
target_compile_options(kaiinfer_core PRIVATE -fblocks $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
if(AGENTARMOR_METAL)
  target_compile_definitions(kaiinfer_core PUBLIC KAI_METAL=1)
  target_link_libraries(kaiinfer_core PUBLIC "-framework Metal")
endif()
add_subdirectory(server)

if(AGENTARMOR_BUILD_FUZZERS)
  if(NOT AGENTARMOR_SANITIZER STREQUAL "address")
    message(FATAL_ERROR "Fuzzers require AGENTARMOR_SANITIZER=address")
  endif()
  execute_process(COMMAND "${CMAKE_CXX_COMPILER}" --print-file-name=libclang_rt.fuzzer_osx.a
    OUTPUT_VARIABLE armor_fuzzer_runtime OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT EXISTS "${armor_fuzzer_runtime}")
    message(FATAL_ERROR "This Clang lacks libFuzzer. Use a full LLVM toolchain; deterministic mutation tests remain available.")
  endif()
  add_executable(armor_protocol_fuzz tests/protocol_fuzz.cpp)
  target_link_libraries(armor_protocol_fuzz PRIVATE kaiinfer_protocol)
  target_compile_options(armor_protocol_fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(armor_protocol_fuzz PRIVATE -fsanitize=fuzzer)
endif()
